#include <string>
#include <vector>
#include <iostream>

using namespace std;

int solution(vector<vector<int>> board, int k) {
    int answer = 0;
    int x = board.size();
    int y = board[0].size();
    if (x < 100 && y < 100 && 0 <= k && k < x + y) {
        for (int i = 0; i < x; i++) {
            for (int j = 0; j < y; j++) {
                if (i + j <= k) {
                    answer += board[i][j];
                }
            }
        }
    }
    return answer;
}
